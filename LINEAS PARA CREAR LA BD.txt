LINEAS PARA CREAR LA BD

CREATE DATABASE veterinaria;

CREATE TABLE usuario(
    Ci VARCHAR(8) PRIMARY KEY,
    Nombre VARCHAR(100),
    Apellido VARCHAR(100),
    Direccion VARCHAR(100),
    Mail VARCHAR(250),
    Telefono VARCHAR(20),
    Rol VARCHAR(30)
);

CREATE TABLE Login(
    NombreDeUsuario VARCHAR(50) PRIMARY KEY,
    Contraseña VARCHAR(250),
    CiUsuario VARCHAR(8)
);

ALTER TABLE login
ADD CONSTRAINT fk_CiUsuario
FOREIGN KEY (CiUsuario)
REFERENCES usuario(Ci)
ON UPDATE CASCADE
ON DELETE CASCADE;


CREATE TABLE Socio ( IdMembresia INT PRIMARY KEY AUTO_INCREMENT, fechaDePago DATE, FechaDeCierre DATE, CiUsuario VARCHAR(8), CONSTRAINT fk_CiUsuarioSocio FOREIGN KEY (CiUsuario) REFERENCES usuario(Ci) ON UPDATE CASCADE ON DELETE CASCADE );


CREATE TABLE Mascota (
    Id INT PRIMARY KEY AUTO_INCREMENT,
    Nombre VARCHAR(100),
    Raza VARCHAR(50),
    FechaNacimiento DATE,
    Castrado BOOLEAN,
    Peso FLOAT,
    CiUsuario VARCHAR(8),
    CONSTRAINT fk_CiUsuarioAnimal
        FOREIGN KEY (CiUsuario)
        REFERENCES usuario(Ci)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE TABLE proveedor(
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(75),
    Telefono VARCHAR(25)
);

CREATE TABLE producto(
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Descripcion VARCHAR(255),
    Precio DECIMAL(10,2) NOT NULL,
    Stock INT NOT NULL,
    Tipo VARCHAR(50),
    Marca VARCHAR(50),
    FechaVencimiento DATE,
    IdProveedor INT,

    CONSTRAINT fk_IdProveedor
        FOREIGN KEY (IdProveedor)
        REFERENCES proveedor(Id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE imagen (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Ruta VARCHAR(500) NOT NULL,
    Tipo VARCHAR(50),
    FechaSubida DATETIME DEFAULT CURRENT_TIMESTAMP,
    IdProducto INT NOT NULL,

    CONSTRAINT fk_ImagenProducto
        FOREIGN KEY (IdProducto)
        REFERENCES producto(Id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE TABLE factura(
      IdFactura INT AUTO_INCREMENT PRIMARY KEY,
    Fecha date,
    PrecioFinal float,
    TipoDePago varchar(30),
    Hora time,
    NombreEmpresa varchar(100),
    RUT varchar(20)


    );
